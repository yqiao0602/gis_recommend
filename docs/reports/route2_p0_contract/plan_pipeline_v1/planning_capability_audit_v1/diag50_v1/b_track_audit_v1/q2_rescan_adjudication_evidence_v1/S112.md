function bitwiseExtract(value, fromBit, toBit) {

 if (toBit === undefined) toBit = fromBit

 var mS = ee.Number(1).add(toBit).subtract(fromBit)

 var mK = ee.Number(1).leftShift(mS).subtract(1)

 return value.rightShift(fromBit).bitwiseAnd(mK)

}

function cloudfree_landsat7 (image){

 var qa = image.select('pixel_qa')

 var clS = bitwiseExtract(qa, 5)

 var clSS = bitwiseExtract(qa, 3)

 var mK = clS.eq(0) // 清晰

 .and(clSS.eq(0)) //无云影

 return image.updateMask(mK)

}

function NDVI(img) {

 var ndvi = img.normalizedDifference(["B5","B4"]);

 return img.addBands(ndvi)

}

function NDWI(img){

 var ndwi = img.normalizedDifference(["B3","B5"]);

 return img.addBands(ndwi)

}

var image = ee.ImageCollection(landsat5).filterDate("2006-01-01","2006-12-31")

              .filterBounds(jlj);

var image1 = image.map(cloudfree_landsat7).mosaic();

var clipped = image1.clip(jlj);

var visualization = {

  min:0,

  max:3500,

  bands: ['B3', 'B2', 'B1'],

};

Map.centerObject(jlj,8)

Map.addLayer(clipped,visualization, 'L7');

clipped = NDVI(clipped);

clipped = NDWI(clipped);

print(clipped)

var newfc = water.merge(vegetation).merge(urban)

              .merge(paddy).merge(farm).merge(unused).merge(beach);

var bands=['B2','B3','B4','B5','B6','B7','nd','nd_1'];

var training = clipped.select(bands).sampleRegions({

 collection:newfc,

 properties:['type'],

 scale:30

});

var withRandom = training.randomColumn('random');

var spliting = 0.5;//Roughly 80% training,20% testing

var trainingPartition = withRandom.filter(ee.Filter.lt('random',spliting));

var testingPartition = withRandom.filter(ee.Filter.gte('random',spliting));

var classifier = ee.Classifier.libsvm().train({

 features: trainingPartition,

 classProperty: 'type',

 inputProperties: bands

});

var test = testingPartition.classify(classifier);

var classified= clipped.select(bands).classify(classifier);

Map.addLayer(classified, {min: 0, max: 6, palette: ['blue', 'green', 'orange','yellow','#00ffff','red','pink']});

var confusionMatrix = test.errorMatrix('type', 'classification');//计算混淆矩阵

print('confusionMatrix',confusionMatrix);//输出显示混淆矩阵

print('consumers accuracy',confusionMatrix.consumersAccuracy());//输出显示用户精度

print('producers accuracy',confusionMatrix.producersAccuracy());//输出显示制作精度

print('overall accuracy', confusionMatrix.accuracy());//输出显示总体精度

print('kappa accuracy', confusionMatrix.kappa());//输出显示 Kappa 系数

Export.image.toDrive({

 image: classified,

 description: '2006jlj',

 folder: 'bysj',

 scale: 30,//分辨率

 region: jlj,

 maxPixels:34e10

 })
