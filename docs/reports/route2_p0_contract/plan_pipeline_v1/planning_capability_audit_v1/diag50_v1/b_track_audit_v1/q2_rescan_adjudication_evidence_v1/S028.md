    var bandsRF = ['B2', 'B3', 'B4', 'B5', 'B6', 'NDVI', 'NDBI', 'medianEVI2', 'ampEVI2','entropyNDBI'];

    var referencemap = amostraTotal

        reducer: ee.Reducer.first()

      })

        "format": "png"

    },  'CLASS_'+i_ano,false);

    Export.image.toDrive({

        image: classified_RF,

        description: "CLASS_RF_DRIVE"+i_ano+class_tipo,

        scale: 10,

        fileFormat: 'GEOTIFF',

        maxPixels: 1e13,

        region: area_estudo,

    });
