function maskS2clouds(image) {

  var qa = image.select('QA60');

  var cloudBitMask = 1 << 10;

  var cirrusBitMask = 1 << 11;

  var mask = qa.bitwiseAnd(cloudBitMask).eq(0)

      .and(qa.bitwiseAnd(cirrusBitMask).eq(0));

  return image.updateMask(mask).divide(10000).copyProperties(image, ["system:time_start"]);

}

var coleccion = ee.ImageCollection('COPERNICUS/S2_SR')

  .filterBounds(parcela) //Filtramos sólo para la parcela trigo

  .filterDate('2017-09-01','2018-08-31')

var coleccion_sin_nubes=coleccion.map(maskS2clouds);

print(coleccion_sin_nubes);

print(coleccion);

function NDMI(entrada){

var ndmi = entrada.addBands(entrada.normalizedDifference(['B8','B11'])

.rename('NDMI'))

return (ndmi)

}

function NDVI(entrada){

  var ndvi = entrada.normalizedDifference(['B8','B4']);

  return entrada.addBands(ndvi.rename('NDVI'));

}

function CIG(entrada){

  var cig =entrada.expression('(NIR/GREEN)-1',

  { NIR: entrada.select ("B8"),

    GREEN: entrada.select ("B3")  })

  .rename('CIG');

  return (cig).copyProperties(entrada, ["system:time_start"]);

}

Map.centerObject(parcela,15);

Map.addLayer(parcela,{color:'5A63CA',stroleWidth:500},"Parcelas",false);

var coleccion_ndvi = coleccion_sin_nubes.map(NDVI);

print(coleccion_ndvi)

var girasol_parcela = parcela.filter(ee.Filter.eq('cultivo','Girasol'));

Map.addLayer(girasol_parcela,{color:'4FDE3B',stroleWidth:500},"Girasol",false);

Map.centerObject(girasol_parcela,13);

print(girasol_parcela);

var trigo_parcela = parcela.filter(ee.Filter.eq('cultivo','Trigo'));

Map.addLayer(girasol_parcela,{color:'F2BA42',stroleWidth:500},"Trigo",false);

Map.centerObject(trigo_parcela,13);

print(trigo_parcela);

var coleccion_ndvi_cig_ndmi = coleccion_sin_nubes.map(function(image){

  var ndvi = NDVI(image);

  var cig = CIG(image);

  var ndmi = NDMI(image);

  return ndvi.addBands([cig, ndmi]);

});

print(coleccion_ndvi_cig_ndmi);

var coleccion_girasol_ndvi_cig_ndmi = coleccion_ndvi_cig_ndmi

  .filterBounds(girasol_parcela)

  .select(['NDVI', 'CIG', 'NDMI']);

print(coleccion_girasol_ndvi_cig_ndmi);

var serie_media=ui.Chart.image.series({

  imageCollection: coleccion_girasol_ndvi_cig_ndmi,

  region: girasol_parcela,

  reducer: ee.Reducer.mean(),

  scale: 10,

  xProperty: 'system:time_start'

}).setOptions({

  title: 'Evolución temporal de NDVI, CIG y NDMI en parcelas de girasol',

  vAxis: {title: 'Valor medio'},

  hAxis: {title: 'Fecha'},

  lineWidth: 1,

  pointSize: 2,

});

print(serie_media);

var coleccion_trigo_ndvi_cig_ndmi = coleccion_ndvi_cig_ndmi

  .filterBounds(trigo_parcela)

  .select(['NDVI', 'CIG', 'NDMI']);
