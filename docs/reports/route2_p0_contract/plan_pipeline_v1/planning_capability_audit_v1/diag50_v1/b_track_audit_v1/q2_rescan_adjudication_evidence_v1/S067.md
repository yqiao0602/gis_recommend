var doLandsat8 = function() {

  var Wcollection = ee.ImageCollection("LANDSAT/LC08/C01/T1_SR")

    .filterBounds(StudyArea)

    .filterDate(start_Winter, end_Winter);

  var Scollection = ee.ImageCollection("LANDSAT/LC08/C01/T1_SR")

     .filterBounds(StudyArea)

     .filterDate(start_Spring, end_Spring);

  var addDate = function(image) {

     var date = ee.Date(image.get('system:time_start'));

     var dateString = date.format('YY-MMdd');

    return image.set('date', dateString);

  };

  var W_dated = Wcollection.map(addDate);

  var S_dated = Scollection.map(addDate);

  function maskCloud(img) {

    var clear = img.select('pixel_qa').bitwiseAnd(2).neq(0);

    return img.updateMask(clear);}

  var W_cloudmasked = W_dated.map(maskCloud);

  var S_cloudmasked = S_dated.map(maskCloud);

  var LS8_getNDVI = function(image) {

    return image.addBands(image.normalizedDifference(['B5', 'B4']));

  };

  var NDVI_W = W_cloudmasked.map(LS8_getNDVI);

  var NDVI_S = S_cloudmasked.map(LS8_getNDVI);

  var NDVI_clipped = NDVI_S.map(

    function(image) {

      return image.clip(workingData);

  });

  NDVI_W = NDVI_W.map(function(i) {

    return ee.Image(i)

      .select(['nd'],[ee.Date(ee.Image(i).get('system:time_start')).format('YY-MMdd')])});

  NDVI_S = NDVI_S.map(function(i) {

    return ee.Image(i)

      .select(['nd'],[ee.Date(ee.Image(i).get('system:time_start')).format('YY-MMdd')])});

    return ee.Image(i2).addBands(ee.Image(i1));

  var S_STACK = NDVI_S.iterate(stack, ee.Image());

      .slice(1)

      .reduceRegion({

        reducer: 'mean',

        geometry: feature.geometry(),

        scale: 30,

        maxPixels: 1e9

      }).map(function(key, value) {

        return ee.Algorithms.If(value, value, 0);

      }));

  };

  var Wresults = workingData.map(ww);

    return feature.set(ee.Image(S_STACK)

      .reduceRegion({

        reducer: 'mean',

        geometry: feature.geometry(),

        maxPixels: 1e9

      }).map(function(key, value) {

        return ee.Algorithms.If(value, value, 0) ;

      }));

  };

  var Sresults = workingData.map(ss);

  return [Wresults, Sresults, NDVI_W, NDVI_S];

};

var extractNDVI = function(field) {

  var fieldDict = field.toDictionary();

  var fieldKeys = fieldDict.keys();

  var fieldVals = fieldDict.values();

  var ndviFn = function(key, val) {

    key = ee.String(key);

    var keyIndex = ee.Number(key.index('-'));

    return ee.Algorithms.If(ee.Algorithms.IsEqual(keyIndex, -1), null, val);

  };

  var ndviDict = fieldDict.map(ndviFn);

  fieldKeys = ndviDict.keys();

  fieldVals = ndviDict.values();

  var fieldVals2 = fieldVals;

  fieldVals2 = fieldVals.sort();

  var biggest = fieldVals2.get(-1);

  var biggestIndex = fieldVals.indexOf(biggest);

  var maxDate = fieldKeys.get(biggestIndex);

  fieldDict = fieldDict.set('Max_Date_NEW', maxDate);

  fieldDict = ee.Dictionary(fieldDict);

  fieldDict = fieldDict.set('Max_NDVI_NEW', biggest);

  fieldDict = ee.Dictionary(fieldDict);

  fieldDict = fieldDict.set('Satellite', satKey);

  fieldDict = ee.Dictionary(fieldDict);

  fieldDict = fieldDict.set('Season', season);

  var newField = field.set(fieldDict);

  return newField;

};

var convertNDVI = function(field) {

  var ndvi = ee.Number(field.get('Max_NDVI_NEW'));

  var bioLog = ndvi.multiply(Number(bioSlopeFinal)).add(Number(bioIntFinal));

  bioLog = bioLog.add(bioRMSEFinal / 2);

  var biomass = bioLog.exp();

  var pgc = ndvi.multiply(Number(pgcSlopeFinal)).add(Number(pgcIntFinal));

  var newField = field.set('Biomass', biomass);

  newField = newField.set('Percent Ground Cover', pgc);

  return newField;

};

var getMoney = function(field) {

  var amount = ee.Number(field.get('Amount'));

  var biomass = ee.Number(field.get('Biomass'));

  var money = amount.multiply(biomass);

  money = money.divide(50);

  field = field.set('DollarAmt', money);

  return field;

};

      Map.remove(layers);

      var octDates = workingData.filter(ee.Filter.eq('PlantMonth', 10));

      var novDates = workingData.filter(ee.Filter.eq('PlantMonth', 11));

      octDates = octDates.filter(ee.Filter.gt('PlantDay', 15));

      workingData = octDates.merge(novDates);

      layers = Map.addLayer(workingData, {}, 'New layer');

    }

  }

});

var doLandsat8 = function() {

  var Wcollection = ee.ImageCollection("LANDSAT/LC08/C01/T1_SR")

    .filterBounds(workingData)

    .filterDate(start_Winter, end_Winter);

  var Scollection = ee.ImageCollection("LANDSAT/LC08/C01/T1_SR")

     .filterBounds(workingData)

     .filterDate(start_Spring, end_Spring);

  Map.addLayer(Wcollection.filterBounds(workingData),

    {'bands': ['B4', 'B3', 'B2'], 'min': 1000, 'max': 9500}, 'LS8', false);

  var addDate = function(image) {

     var date = ee.Date(image.get('system:time_start'));

     var dateString = date.format('YY-MMdd');

    return image.set('date', dateString);

  };

  var W_dated = Wcollection.map(addDate);

  var S_dated = Scollection.map(addDate);

  function maskCloud(img) {

    var clear = img.select('pixel_qa').bitwiseAnd(2).neq(0);

    return img.updateMask(clear);}

  var W_cloudmasked = Wcollection.map(maskCloud);

  var S_cloudmasked = Scollection.map(maskCloud);

  var LS8_getNDVI = function(image) {

    return image.addBands(image.normalizedDifference(['B5', 'B4']));

  };

  var NDVI_W = W_cloudmasked.map(LS8_getNDVI);

  var NDVI_S = S_cloudmasked.map(LS8_getNDVI);

  var NDVI_clipped = NDVI_S.map(

    function(image) {

      return image.clip(workingData);

  });

  Map.addLayer(NDVI_clipped, {bands: ['nd'], palette: '000000, 00FF00', min: 0, max: 0.7}, 'NDVI');

  NDVI_W = NDVI_W.map(function(i) {

    return ee.Image(i)

      .select(['nd'],[ee.Date(ee.Image(i).get('system:time_start')).format('YY-MMdd')])});

  NDVI_S = NDVI_S.map(function(i) {

    return ee.Image(i)

      .select(['nd'],[ee.Date(ee.Image(i).get('system:time_start')).format('YY-MMdd')])});

  var stack = function (i1, i2) {

    return ee.Image(i2).addBands(ee.Image(i1));

  };

  var W_STACK = NDVI_W.iterate(stack, ee.Image());

  var S_STACK = NDVI_S.iterate(stack, ee.Image());

  var ww = function(feature) {

    return feature.set(ee.Image(W_STACK)

      .slice(1)

      .reduceRegion({

        reducer: 'mean',

        geometry: feature.geometry(),

        scale: 30,

        maxPixels: 1e9

      }).map(function(key, value) {

        return ee.Algorithms.If(value, value, 0);

      }));

  };

  var Wresults = workingData.map(ww);

  var ss = function(feature) {

    return feature.set(ee.Image(S_STACK)

      .reduceRegion({

        reducer: 'mean',

        geometry: feature.geometry(),

        scale: 30,

        maxPixels: 1e9

      }).map(function(key, value) {

        return ee.Algorithms.If(value, value, 0) ;

      }));

  };

  var Sresults = workingData.map(ss);

  return [Wresults, Sresults];

};
