function maskL8sr(image) {

  var cloudShadowBitMask = 1 << 3;

  var cloudsBitMask = 1 << 5;

  var qa = image.select('pixel_qa');

  var mask = qa.bitwiseAnd(cloudShadowBitMask).eq(0)

      .and(qa.bitwiseAnd(cloudsBitMask).eq(0));

  return image.updateMask(mask).divide(10000)

      .select("B[0-9]*")

      .copyProperties(image, ["system:time_start"]);

}

var addNDVI = function(image) {

  var ndvi = image.normalizedDifference(['B5', 'B4']).rename('NDVI');

  return image.addBands(ndvi);

};

var ROI=ee.FeatureCollection('users/Global_fire/Changbaishan')

var image2014= ee.ImageCollection('LANDSAT/LC08/C01/T1_SR')

              .filterDate('2017-06-01', '2017-10-01')

              .filterBounds(ROI)

              .sort('CLOUD_COVER')

              .map(maskL8sr);

var image2015= ee.ImageCollection('LANDSAT/LC08/C01/T1_SR')

              .filterDate('2015-06-01', '2015-10-01')

              .filterBounds(ROI)

              .sort('CLOUD_COVER')

              .map(maskL8sr);

var image=image2015.merge(image2014)

print(image)

print(NDVI)

function calFVC(BestVI,region,scale){

    var num = BestVI.reduceRegion({

      reducer:ee.Reducer.percentile([5,95]),

      geometry:region,

      scale:scale,

      maxPixels:1e13

    });

    var min = ee.Number(num.get("NDVI_p5"));

    var max = ee.Number(num.get("NDVI_p95"));

    var greaterPart = BestVI.gt(max);

    var lessPart = BestVI.lt(min);

    var middlePart =ee.Image(1).subtract(greaterPart).subtract(lessPart);

    var tempf1=BestVI.subtract(min).divide(max.subtract(min));

    var FVC=ee.Image(1).multiply(greaterPart).add(ee.Image(0).multiply(lessPart))

                      .add(tempf1.multiply(middlePart))

    return FVC.rename('FVC');

}

var FVC=calFVC(NDVI,ROI,30)

print(FVC)

print(chart)

});
