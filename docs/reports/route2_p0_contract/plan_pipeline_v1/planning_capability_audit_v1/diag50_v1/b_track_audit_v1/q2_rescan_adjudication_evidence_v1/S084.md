Task: Analyze CO concentration
Analyze carbon monoxide (CO) column number density over DKI Jakarta using Sentinel-5P NRTI L3 data from March 2019. It computes the mean CO concentration, clips it to the DKI Jakarta administrative boundary, visualizes the result with a custom color palette, generates a time-series chart of average CO values by region, and exports the clipped mean image as a GeoTIFF to Google Drive. A legend is also added to the map for visualization reference.

## Instruction B（全文）

Task: Analyze CO concentration over time
Analyze carbon monoxide (CO) column number density over DKI Jakarta in March 2019 using Sentinel-5P satellite data. It computes the mean CO concentration across the region, clips it to the DKI administrative boundary, visualizes the result with a custom color palette, generates a time-series chart of average CO values by date for each sub-region, and exports the mean raster as a GeoTIFF to Google Drive. A legend is also added to the map for visual interpretation.

## Source A（全文）

```javascript
Map.setCenter(106.84, -6.21, 4);

var DKIBorder = ee.FeatureCollection('users/sachiemelin/WilayahAdministrasi/DKI');

var geom = ee.Geometry.Point(106.84, -6.21).buffer(10000);

var DKIBorder = ee.FeatureCollection('users/sachiemelin/WilayahAdministrasi/DKI');

print(DKIBorder);

Map.centerObject(DKIBorder, 6);

Map.addLayer(DKIBorder);

var collection = ee.ImageCollection('COPERNICUS/S5P/NRTI/L3_CO')

  .select('CO_column_number_density')

  .filterDate('2019-03-01', '2019-03-31');

var band_viz = {

  min: 0,

  max: 0.05,

  palette: ['black', 'blue', 'purple', 'cyan', 'green', 'yellow', 'red']

};

Map.addLayer(collection.mean().clip(DKIBorder), band_viz, 'S5P CO');

var palette = palette;

var vis = band_viz;

Map.addLayer(collection.mean().clip(DKIBorder), band_viz, 'S5P CO 2019');

function makeColorBarParams(palette) {

  return {

    bbox: [0, 0, 1, 0.1],

    dimensions: '100x10',

    format: 'png',

    min: 0,

    max: 1,

    palette: palette,

  };

}

var colorBar = ui.Thumbnail({

  image: ee.Image.pixelLonLat().select(0),

  params: makeColorBarParams(vis.palette),

  style: {stretch: 'horizontal', margin: '0px 8px', maxHeight: '24px'},

});

var legendLabels = ui.Panel({

  widgets: [

    ui.Label(vis.min, {margin: '4px 8px'}),

    ui.Label(

        ((vis.max-vis.min) / 2+vis.min),

        {margin: '4px 8px', textAlign: 'center', stretch: 'horizontal'}),

    ui.Label(vis.max, {margin: '4px 8px'})

  ],

  layout: ui.Panel.Layout.flow('horizontal')

});

var legendTitle = ui.Label({

  value: 'CO',

  style: {fontWeight: 'bold'}

});

var legendPanel = ui.Panel([legendTitle, colorBar, legendLabels]);

Map.add(legendPanel);

var chart =

    ui.Chart.image

        .seriesByRegion({

          imageCollection: collection,

          regions: DKIBorder,

          reducer: ee.Reducer.mean(),

          scale: 500,

          seriesProperty: 'label',

          xProperty: 'system:time_start'

        })

        .setOptions({

          title: 'Average CO Value by Date',

          hAxis: {title: 'Date', titleTextStyle: {italic: false, bold: true}},

          vAxis: {

            title: 'CO ',

            titleTextStyle: {italic: false, bold: true}

          },

          lineWidth: 5,

          colors: ['f0af07', '0f8755', '76b349'],

        });

print(chart);

Export.image.toDrive({

  image: collection.mean().clip(DKIBorder),

  description: 'CO_DKI_2019',

  scale: 100,

  region: DKIBorder,

  fileFormat: 'GeoTIFF',

  formatOptions: {

    cloudOptimized: true

  }

});

```
