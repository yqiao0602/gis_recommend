## Instruction A（全文）

Task: Analyze methane concentration levels
Analyze the column-averaged dry air mixing ratio of methane (CH4) over North India using Sentinel-5P satellite observations from May 2019 to April 2020. It applies cloud masking based on a 25% cloud fraction threshold, computes the mean CH4 concentration across the time period, clips the result to a user-defined rectangular region and to India's national boundary, generates a time-series chart of regional CH4 levels, visualizes the mean concentration with a custom color palette, adds an interactive legend to the map, and exports the clipped mean raster and country boundary as Drive assets. Outputs include a classified raster map of average CH4 levels, a time-series chart, and exported KMZ and GeoTIFF files for external use.

## Instruction B（全文）

Task: Analyze methane concentration levels
Analyze the column-averaged dry air mixing ratio of methane (CH4) over Brazil using Sentinel-5P satellite observations from May 2019 to April 2020. It applies cloud masking based on a user-defined cloud fraction threshold (0.25), computes the mean CH4 concentration across the time period, clips the result to the country boundary, generates a time series chart of regional CH4 levels, visualizes the data with a color palette, adds an interactive legend to the map, and exports the mean CH4 raster and country border as Drive assets. Outputs include a classified raster map of average methane concentrations, a time-series chart, and exported KMZ and GeoTIFF files for external use.

var region_country = ee.FeatureCollection('USDOS/LSIB_SIMPLE/2017')

                    .filterMetadata('country_co', 'equals', 'IN');

var region = region_country; ///(make region equal to your country of interest)

var regionborders = region;

var scales = 250;  ///enter a scale (m/pix)

var start = ee.Date('2019-05-01'); ///enter a start date of observations

var finish = ee.Date('2020-04-30'); /// enter a finish date of observations

region = ee.FeatureCollection(ee.Geometry.Rectangle(73.00, 26.00, 79.00, 32.00) );

scales = 1000;

var cloudFraction = 0.25; ///percents; value between 0 - 1

var cloudMask = function(image) {

      return image.updateMask(image.select("cloud_fraction").lt(cloudFraction));

    };

var CH4=ee.ImageCollection("COPERNICUS/S5P/OFFL/L3_CH4")

 .select('CH4_column_volume_mixing_ratio_dry_air')

 .filterDate(start, finish);

var band_viz_CH4 = {

  min: 1800,

  max: 1950,

  palette: [ '00008f','0003ff', '03fffc', 'fffc00', 'ff0300', '830000']

};

var CH4mean = CH4.reduce(ee.Reducer.mean());

var CH4meanclipped = CH4mean.clip(region);

var CH4meancountry = CH4mean.clip(region_country);

var CH4Chart = ui.Chart.image.seriesByRegion( CH4, region, ee.Reducer.mean(), 'CH4_column_volume_mixing_ratio_dry_air', scales, 'system:time_start','CH4');

print(CH4Chart);

Map.addLayer(CH4meancountry.select([0]), band_viz_CH4, 'S5P CH4 Country', true);

Map.addLayer(CH4meanclipped.select([0]), band_viz_CH4, 'S5P CH4 Feature', true);

var legend = ui.Panel({

  style: {

    position: 'bottom-left',

    padding: '8px 10px'

  }

});

 var legendTitle = ui.Label({

  value: ('Column averaged dry air \nmixing ratio of methane\n (ppbV)'),

  style: {

    fontWeight: 'bold',

    fontSize: '15px',

    margin: '0 0 4px 0',

    padding: '0',

    whiteSpace:'pre',

    }

});

var gradient = lon.multiply((band_viz_CH4.max-band_viz_CH4.min)/100.0).add(band_viz_CH4.min);

var legendImage = gradient.visualize(band_viz_CH4);

var panel = ui.Panel({

    widgets: [

      ui.Label(band_viz_CH4['max'])

    ],

  });

legend.add(panel);

var thumbnail = ui.Thumbnail({

  image: legendImage,

  params: {bbox:'0,0,10,100', dimensions:'10x200'},

  style: {padding: '1px', position: 'bottom-center'}

});

legend.add(thumbnail);

var panel = ui.Panel({

    widgets: [

      ui.Label(band_viz_CH4['min'])

    ],

  });

legend.add(panel);

Map.add(legend);

var palette = [ '00008f','0003ff', '03fffc', 'fffc00', 'ff0300', '830000'];

var CH4vis = {min: 1800, max: 1950, palette: palette};

var toexport5 = CH4meanclipped.visualize(CH4vis);

Map.centerObject(region);

Export.image.toDrive({

  region: region,
