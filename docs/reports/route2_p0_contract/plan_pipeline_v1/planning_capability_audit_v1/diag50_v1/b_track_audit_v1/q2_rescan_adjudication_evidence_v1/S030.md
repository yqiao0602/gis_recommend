  function create_superpixel(dataset) {

    var winter = dataset.filter(ee.Filter.calendarRange(11, 2, 'month'))  // Winter

    .sort('CLOUD_COVER', false)   // Sort by cloud cover

    .mosaic()   // create a mosaic

    .select(['R', 'NIR', 'SWIR']);  // band combination

    var footprint = dataset.geometry().bounds();

    var grid_size = 10;

    var seeds = ee.Algorithms.Image.Segmentation.seedGrid(grid_size);

    var snic = ee.Algorithms.Image.Segmentation.SNIC({

      image: winter,

      compactness: 0,

      connectivity: 8,

      neighborhoodSize:256,

      seeds: seeds

    });

    var clusters = snic.select('clusters');

    var std = winter.addBands(clusters)

    .reduceConnectedComponents(ee.Reducer.variance(), 'clusters', 256)

    .reduce('sum');

    var std_mu = std.updateMask(seeds)

    .reduceRegion({

      reducer: ee.Reducer.median(),

      geometry: footprint,

      scale: 30*grid_size,

      tileScale: 2

    })

    .get('sum');

    var outliers = std.gt(ee.Number(std_mu).multiply(1.0));

    var new_seeds = ee.Algorithms.Image.Segmentation.seedGrid(grid_size/2).updateMask(outliers);

    seeds = seeds.unmask().or(new_seeds.unmask());

    var snic2 = ee.Algorithms.Image.Segmentation.SNIC({

      image: winter,

      compactness: 0,

      connectivity: 8,

      neighborhoodSize:256,

      seeds: seeds

    });

    var clusters2 = snic2.select('clusters');

    return clusters2;

  }
