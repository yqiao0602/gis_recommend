Task: Classify land cover types
Classify land cover types in Cumberland County, Maine using Landsat 8 surface reflectance imagery from summer 2019 and 2020. It applies cloud and shadow masking, computes a median composite, and trains a Random Forest classifier on labeled training points for eight classes: low/mid/high density development, coniferous, mixed forest, deciduous, cultivated, and water. The model is validated with test data, producing accuracy metrics including confusion matrix, overall accuracy, and kappa coefficient. The final classification is merged with NLCD impervious surface data to refine urban areas and displayed as a map with a legend and SLD-styled visualization.

function maskL8sr(image) {

  var cloudShadowBitMask = 1 << 3;

  var cloudsBitMask = 1 << 5;

  var qa = image.select('pixel_qa');

  var mask = qa.bitwiseAnd(cloudShadowBitMask).eq(0)

      .and(qa.bitwiseAnd(cloudsBitMask).eq(0));

  return image.updateMask(mask).divide(10000)

      .select("B[0-9]*")

      .copyProperties(image, ["system:time_start"]);

}

var sum20 = ee.Filter.date('2020-06-01','2020-09-30');

var sum19 = ee.Filter.date('2019-06-01','2019-09-30');

var SumFilter = ee.Filter.or(sum20, sum19);

var allsum = image.filter(SumFilter);
