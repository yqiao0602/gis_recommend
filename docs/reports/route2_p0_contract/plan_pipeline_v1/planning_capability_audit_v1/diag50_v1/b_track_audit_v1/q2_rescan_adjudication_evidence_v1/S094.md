var image = ee.Image(ee.ImageCollection('LANDSAT/LC08/C01/T1_SR')

    .filterBounds(poi)

    .filterDate('2019-10-01', '2019-11-30')

    .sort('CLOUD_COVER')

    .first());

var training_points = air.merge(urban).merge(vegetasi).merge(tanah).merge(tegalan);

var bands = ['B2', 'B3', 'B4', 'B5', 'B6', 'B7'];

var label = 'LC';

var sampling = image.sampleRegions({

  collection: training_points,

  properties: ['LC'],

  scale:30

});

print("Sampling ", sampling);

var trained = ee.Classifier.libsvm().train(sampling, label, bands);

var classified = image.select(bands).classify(trained);

Map.centerObject(poi, 11);

Map.addLayer(image, {bands: ['B4', 'B3', 'B2'], min:0, max: 3000}, 'image');

Map.addLayer(classified,

             {min: 1, max: 5, palette: ['blue', 'red', 'green','yellow','black']},

             'classification');

var validation = classified.sampleRegions({

  collection: cek_acc,

  properties: ['LC'],

  scale: 30,

});

print('Validation error matrix: ', trained.confusionMatrix());

var testAccuracy = validation.errorMatrix('LC', 'classification');

print('Validation error matrix: ', testAccuracy);

print('Validation overall accuracy: ', testAccuracy.accuracy());
