var Zhang_Suen = function(image, iterations, prj) {

  var knl = ee.Kernel.fixed(3, 3, [[1, 1, 1], [1, 0, 1], [1, 1, 1]]);

  var output = image;

  for(var i=0; i<iterations; i++) {

    var ngb1 = output.neighborhoodToBands(knl).select(['b1_-1_-1','b1_0_-1','b1_1_-1','b1_-1_0',

                        'b1_1_0','b1_-1_1','b1_0_1','b1_1_1'], ['p9','p2','p3','p8','p4','p7','p6','p5']);

    var con11 = ngb1.expression("b('p2')+b('p3')+b('p4')+b('p5')+b('p6')+b('p7')+b('p8')+b('p9')");

        con11 = con11.gte(2).and(con11.lte(6));

    var con12 = ngb1.expression("((b('p2')-b('p3'))>0) + ((b('p3')-b('p4'))>0) + ((b('p4')-b('p5'))>0) + ((b('p5')-b('p6'))>0) + ((b('p6')-b('p7'))>0) + ((b('p7')-b('p8'))>0) + ((b('p8')-b('p9'))>0) + ((b('p9')-b('p2'))>0)");

        con12 = con12.eq(1);

    var con13 = ngb1.expression("b('p2')*b('p4')*b('p6')").eq(0);

    var con14 = ngb1.expression("b('p4')*b('p6')*b('p8')").eq(0);

    output = output.updateMask(con11.and(con12).and(con13).and(con14).eq(0)).unmask(0);

    var ngb2 = output.neighborhoodToBands(knl).select(['b1_-1_-1','b1_0_-1','b1_1_-1','b1_-1_0',

                        'b1_1_0','b1_-1_1','b1_0_1','b1_1_1'], ['p9','p2','p3','p8','p4','p7','p6','p5']);

    var con21 = ngb2.expression("b('p2')+b('p3')+b('p4')+b('p5')+b('p6')+b('p7')+b('p8')+b('p9')");

        con21 = con21.gte(2).and(con21.lte(6));

    var con22 = ngb2.expression("((b('p2')-b('p3'))>0) + ((b('p3')-b('p4'))>0) + ((b('p4')-b('p5'))>0) + ((b('p5')-b('p6'))>0) + ((b('p6')-b('p7'))>0) + ((b('p7')-b('p8'))>0) + ((b('p8')-b('p9'))>0) + ((b('p9')-b('p2'))>0)");

        con22 = con22.eq(1);

    var con23 = ngb2.expression("b('p2')*b('p4')*b('p8')").eq(0);

    var con24 = ngb2.expression("b('p2')*b('p6')*b('p8')").eq(0);

    output = output.updateMask(con21.and(con22).and(con23).and(con24).eq(0)).unmask(0);

  }

  return output.reproject(prj);

};
