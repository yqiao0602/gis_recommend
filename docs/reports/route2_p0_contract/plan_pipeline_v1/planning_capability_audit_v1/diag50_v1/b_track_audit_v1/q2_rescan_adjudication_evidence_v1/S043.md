Task: Extract NDVI time series from points
Extract NDVI values over a set of point features from Sentinel-2 surface reflectance imagery between April 1 and May 7, 2021. It applies cloud and cirrus masking using the QA60 band, computes multiple vegetation and spectral indices (NDVI, NBR, GNDVI, EVI, CVI, CIgreen, CI, RI, RE1, RE2, RVI, GRVI), and performs region reduction using reduceRegions to extract first non-null NDVI values at each point. The output is a flattened feature collection where each feature represents a point with its corresponding NDVI value and source image ID for time-series analysis.

function maskCloudAndShadows(image)  {

  var qa = image.select('QA60')

  var cloudBitMask = 1 << 10;

  var cirrusBitMask = 1 << 11;

  var mask = qa.bitwiseAnd(cloudBitMask).eq(0).and(

  return image.updateMask(mask).divide(10000)

}

var format = function(table, rowId, colId) {

  var rows = table.distinct(rowId);

  var joined = ee.Join.saveAll('matches').apply({

    primary: rows,

    secondary: table,

    condition: ee.Filter.equals({

      leftField: rowId,

      rightField: rowId

    })

  });

  return joined.map(function(row) {

      var values = ee.List(row.get('matches'))

        .map(function(feature) {

          feature = ee.Feature(feature);

          return [feature.get(colId), feature.get('RI')];

      return row.select([rowId]).set(ee.Dictionary(values.flatten()));
