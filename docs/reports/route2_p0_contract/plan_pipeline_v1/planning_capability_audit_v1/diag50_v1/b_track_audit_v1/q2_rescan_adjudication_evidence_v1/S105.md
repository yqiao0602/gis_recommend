Map.setCenter(46.8319, 30.8545, 11);

var landsatCollection =Landsat7

    .filterDate('2015-05-01','2015-05-31');

var bands = ['B1','B2','B3','B4','B5','B7','B8','elevation'];

var ColorPalette = [

  'blue', // water

  'red', // marsh

  'orange', // bare

  'green' // mix

];

var spectral = ee.Algorithms.Landsat.simpleComposite({

  collection: landsatCollection,

  percentile:25,

  cloudScoreRange:10,

  asFloat: true

});

var composite = spectral.addBands(srtm.select('elevation'))

var classes = water

.merge(mix)

.merge(bare)

.merge(vegatation)

;

var classProperty = 'landcover';

var training = composite.select(bands).sampleRegions({

  collection: classes,

  properties: [classProperty],

  scale: 30

});

var withRandom = training.randomColumn('random');

var split = 0.75;  // Roughly 70% training, 30% testing.

var trainingPartition = withRandom.filter(ee.Filter.lt('random', split));

var testingPartition = withRandom.filter(ee.Filter.gte('random', split));

var classifier = ee.Classifier.cart({

  minSplitPoplulation: 100

})

.train({

  features: trainingPartition,

  classProperty: classProperty,

  inputProperties: bands

});

var classified = composite.classify(classifier);

Map.addLayer(composite, {min: 0, max: 0.3, bands: ['B4','B3','B2']},'False Color');

Map.addLayer(composite, {min: 0, max: 0.3, bands: ['B7','B4','B2']},'Natural-like');

Map.addLayer(classified, {min: 0, max: 3, palette: ColorPalette},'Classification');

var test = testingPartition.classify(classifier);

var confusionMatrix = test.errorMatrix(classProperty, 'classification');

var accuracy = confusionMatrix.accuracy()

var consumersAccuracy = confusionMatrix.consumersAccuracy()

var producersAccuracy = confusionMatrix.producersAccuracy()

var kappa = confusionMatrix.kappa()

print('Overall Accuracy',accuracy)

print('User Accuracy',consumersAccuracy)

print('Producer Accuracy',producersAccuracy)

print('kappa',kappa)

var areaChart = ui.Chart.image.byClass({

  image: ee.Image.pixelArea().addBands(classified),

  classBand: 'classification',

  region: PostHammar,

  scale: 30,

  reducer: ee.Reducer.sum(),

  classLabels: ['Water','Marsh','Bare','Water and Vegetation']

});

print('Area of Each Class (m^2)',areaChart)

Map.addLayer(ee.Image().paint(PostHammar, 0, 3), {}, 'Hammar Perimeter');

var landsatCollectionT =Landsat7

    .filterDate('2016-08-01','2016-08-30');

var spectralT = ee.Algorithms.Landsat.simpleComposite({
