## Instruction A（全文）

Task: Identify rice paddy fields
Identify rice paddy cultivation areas in Qujing for the year 2019 using Landsat 8 surface reflectance data, combining spectral indices (NDVI, EVI, LSWI, MNDWI, NDBI, SAVI, TVI), texture features from GLCM, topographic variables (elevation, slope, aspect, hillshade), and Tasseled Cap transformation (brightness, greenness, wetness). A MaxEnt classifier is trained on labeled samples to classify land cover, with a post-classification rule (slope < 5°) applied to refine rice detection. The final output includes a classified map, accuracy assessment, area statistics, and export of the rice paddy extent.

## Instruction B（全文）

Task: Identify rice paddy fields
Identify rice paddy fields in Qujing for the year 2019 using Landsat 8 surface reflectance data, combining spectral indices (NDVI, EVI, LSWI, MNDWI, NDBI, SAVI, TVI), texture features from GLCM, terrain variables (elevation, slope, aspect, hillshade), and Tasseled Cap transformation (brightness, greenness, wetness). A Random Forest classifier is trained on labeled samples to classify land cover, with post-classification filtering based on low slope to refine paddy detection. Final outputs include a classified land cover map, rice paddy extent, area statistics, and export of paddy field maps.

function maskL8sr(image)

{

	var timeStart = image.get('system:time_start');

  var srImageList = ee.ImageCollection(' LANDSAT/LC08/C01/T1_SR')

                   .filterMetadata('system:time_start','equals',timeStart)

                   .toList(5);

  var cloudShadowBitMask = ee.Number(2).pow(3).int();

  var cloudsBitMask = ee.Number(2).pow(5).int();

  var qa = image.select('pixel_qa');

  var mask = qa.bitwiseAnd(cloudShadowBitMask).eq(0)

      .and(qa.bitwiseAnd(cloudsBitMask).eq(0));

  return image.updateMask(mask);

}

function ND_VI(image,b1,b2,bName)

{

  var VI = image.normalizedDifference([b1,b2]).rename(bName);

  return VI.updateMask(VI.gt(-1).and(VI.lt(1)));

}

function funEVI(image,B1,B2,B3)

{

	var VI = image.expression('2.5 * (nir - red) / (nir + 6 * red - 7.5 * blue + 1)',

    {

      blue: image.select(B1).multiply(0.0001),

      red:  image.select(B2).multiply(0.0001),

      nir:  image.select(B3).multiply(0.0001)

    }).rename('EVI');

  return VI.updateMask(VI.gt(-1).and(VI.lt(1)));

}

function funTVI(image,B1,B2,B3)

{

	var VI = image.expression('0.5*(120*(nir - green) / 200*(red-green))',

    {

      nir:  image.select(B1).multiply(0.0001),

      green:  image.select(B2).multiply(0.0001),

      red:  image.select(B3).multiply(0.0001),

    }).rename('TVI');

  return VI.updateMask(VI.gt(-1).and(VI.lt(1)));

}

function funSAVI(image,B1,B2)

{

	var VI = image.expression( '(1 + L) * float(nir - red)/ (nir + red + L)',

    {

      nir:  image.select(B1).multiply(0.0001),

      red:  image.select(B2).multiply(0.0001),

      L:0.5

    }).rename('SAVI');

  return VI.updateMask(VI.gt(-1).and(VI.lt(1)));

}

function caculateArea(result) {

  var dict = ee.Image.pixelArea()

              .addBands(result)

              .reduceRegion({

                reducer:ee.Reducer.sum().group({

                  groupField:1,

                  groupName:'type',

                }),

                geometry:result.geometry().bounds(),

                scale:30,

                tileScale:2,

                maxPixels:1e13

              });

  var groups = ee.List(dict.get("groups"));

  var typeNames = ee.List(names);

  var featureList = groups.map(function(group){

    group = ee.Dictionary(group);

    var area = ee.Number(group.get("sum"));

    area = area.divide(1000000);

    var type = ee.Number(group.get("type"));

    var f = ee.Feature(null, {

      "type": type,

      "area": area,

      "name": typeNames.get(type.subtract(1))

    });

    return f;

  });

  var areaFCol = ee.FeatureCollection(featureList);

  var totalArea = ee.Number(areaFCol.aggregate_sum("area"));

  areaFCol = areaFCol.map(function(f){

    var _area = ee.Number(f.get("area"));

    f = f.set("percent", _area.divide(totalArea));

    f = f.set("totalArea", totalArea);

    return f;

  });

  print("caculateArea", areaFCol);

}
