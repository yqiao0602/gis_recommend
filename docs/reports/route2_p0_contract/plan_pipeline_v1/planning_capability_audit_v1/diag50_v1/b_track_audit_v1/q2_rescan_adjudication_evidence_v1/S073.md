## Instruction A（全文）

Task: Classify land cover types
Classify land use and land cover (LULC) for the year 2016 using Landsat 8 TOA reflectance data. It computes spectral indices (NDVI, NDBI, NDWI) to support vegetation, built-up, and water discrimination, then applies a Random Forest classifier trained on user-defined ground control points (GCPs) representing six classes: Waterbodies, Forest, Agricultural_land, Builtuparea, Bareground, and Rangeland. The output includes a classified LULC map, accuracy assessment via confusion matrix (overall accuracy, Kappa, producer's/consumer's accuracy), and area calculation in square kilometers per class. The final classified raster is exported to Google Drive.

```javascript
var startDate = '2016-01-01'

var endDate = '2016-05-31'

Map.addLayer(SA, {color:'red'}, 'Study Area');

var image = ee.Image(ee.ImageCollection('LANDSAT/LC08/C01/T1_TOA')  // Filter to get only images under the region of interest.

    .filterBounds(SA)

    .filterDate(startDate, endDate)

    .filterMetadata('CLOUD_COVER','less_than', 10)

    .select('B.*')

    .median());

var ndvi = image.normalizedDifference(['B5', 'B4']).rename(['ndvi']);

var ndbi = image.normalizedDifference(['B6', 'B5']).rename(['ndbi']);

var ndwi = image.normalizedDifference(['B3', 'B5']).rename(['ndwi']);

Map.addLayer(ndvi.clip(SA), {min:0, max:1, palette: ['white', 'green']}, 'NDVI', 0)

Map.addLayer(ndbi.clip(SA), {min:0, max:1, palette: ['white', 'red']}, 'NDBI', 0)

Map.addLayer(ndwi.clip(SA), {min:0, max:1, palette: ['white', 'blue']}, 'NDWI', 0)

var trueColor432 = image.select(['B4', 'B3', 'B2']);

var falseColor543 = image.select(['B5', 'B4', 'B3']);

var trueColor432Vis = { min: 0.0, max: 0.4};

var falseColor543Vis = {min: 0.0, max: 0.4};

Map.centerObject(SA);

Map.addLayer(trueColor432.clip(SA), trueColor432Vis, 'True Color (432)');

Map.addLayer(falseColor543.clip(SA), falseColor543Vis, 'False Color (543)');

var trainingGcp = Waterbodies.merge(Forest).merge(Agricultural_land).merge(Builtuparea).merge(Bareground).merge(Rangeland); //Add or delete the classes

var validationGcp =WaterbodiesT.merge(ForestT).merge(Agricultural_landT).merge(BuiltupareaT).merge(BaregroundT).merge(RangelandT);//Add or delete the classes

var bands=['B2', 'B3', 'B4', 'B5', 'B6', 'B7'];

var training = image.select(bands).sampleRegions(

  {collection: trainingGcp,

  properties: ['landcover'],

  scale:30,

  tileScale: 1});

var classifier = ee.Classifier.smileRandomForest(50)

.train({

  features: training,

  classProperty: 'landcover',

  inputProperties: bands //composite.bandNames()

});

var classified = image.classify(classifier);

Map.addLayer(classified.clip(SA), {min: 0, max: 6, palette: ['#174aec', '#12ab1a', '#fff87b', '#1fff2b', '#c2b4b4', '#eef6f3']}, 'LULC_2016');

var test = classified.sampleRegions({

  collection: validationGcp,

  properties: ['landcover'],

  tileScale: 1,

  scale: 30,

});

var testConfusionMatrix = test.errorMatrix('landcover', 'classification');

print('Confusion Matrix', testConfusionMatrix);

print('Test Accuracy', testConfusionMatrix.accuracy());

print("Consumer's accuracy", testConfusionMatrix.consumersAccuracy());

print("Producer's accuracy", testConfusionMatrix.producersAccuracy());

print('Kappa statistic', testConfusionMatrix.kappa());

for (var i=0;i<6; i++)

{

var Class = classified.eq(i)

var areaClass = Class.multiply(ee.Image.pixelArea())

var area = areaClass.reduceRegion({

  reducer: ee.Reducer.sum(),

  geometry: SA.geometry(),

  scale: 30,

  maxPixels: 1e10

  })

var ClassAreaSqKm = ee.Number(area.get('classification')).divide(1e6)//.round()

print('Area of class=='+i,ClassAreaSqKm)

}

Export.image.toDrive(

      { image: classified.clip(SA),

    description: "LULC_2016",

    scale: 30,

    region:SA,

    maxPixels: 1e10

    });

```
