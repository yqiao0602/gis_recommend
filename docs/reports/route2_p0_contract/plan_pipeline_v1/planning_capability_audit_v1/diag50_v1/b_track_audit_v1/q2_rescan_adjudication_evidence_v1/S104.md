## Source A（全文）

```javascript
var roi = Counties.filterMetadata('NAME','equals','Cass')

Map.addLayer(roi,{},'Cass')

Map.centerObject(roi,11)

var landsat = ee.Image(l8

  .filterDate('2014-05-01', '2014-10-31')

  .filterBounds(roi)

  .sort('CLOUD_COVER')

  .first());

var input = landsat.updateMask(landsat.select('fmask').lte(1))//only gives portions of image with no clouds or snow

print(NASS)

var training = input.addBands(NASS).stratifiedSample({

  region: roi,

  numPoints: 500,

  seed: 0,

  classBand:'b1'

}).filter(ee.Filter.neq('b1', null));

var classifier = ee.Classifier.randomForest(10)

    .train(training, 'b1');

var classified = input.classify(classifier);

var trainAccuracy = classifier.confusionMatrix();

print('Training Recall ', trainAccuracy.consumersAccuracy()); // Recall

print('Training precision: ', trainAccuracy.producersAccuracy()); // Precision

print('Training accuracy: ', trainAccuracy.accuracy()); // Accuracy

var validation = input.addBands(NASS).stratifiedSample({

  region: roi,

  numPoints: 500,

  seed: 1,

  classBand:'b1'

}).filter(ee.Filter.neq('b1', null));  // Filter the result to get rid of any null pixels.

var validated = validation.classify(classifier);

var testAccuracy = validated.errorMatrix('b1', 'classification');

print('Validation Recall ', testAccuracy.consumersAccuracy()); // Recall

print('Validation precision: ', testAccuracy.producersAccuracy()); // Precision

print('Validation accuracy: ', testAccuracy.accuracy()); // Accuracy

classified=classified.select(['classification'],['b1'])

Map.addLayer(landsat,l8vis,'l8')

Map.addLayer(NASS,lcParams,'NASS')

Map.addLayer(classified,lcParams,'LCout')

var classHist=classified.reduceRegion(ee.Reducer.frequencyHistogram().unweighted(),roi.geometry())

var nassHist=NASS.reduceRegion(ee.Reducer.frequencyHistogram().unweighted(),roi.geometry())

print(classHist)

print(nassHist)

var classHistv = ee.FeatureCollection([ee.Feature(null, classHist)]);

var nassHistv = ee.FeatureCollection([ee.Feature(null, nassHist)]); //ground truth

Export.table.toDrive({

  collection: classHistv,

  description: 'classHistv_500_strat_rF_Run2',

  fileFormat: 'CSV'

});

Export.table.toDrive({

  collection: nassHistv,

  description: 'nassHistv',

  fileFormat: 'CSV'

});
```
