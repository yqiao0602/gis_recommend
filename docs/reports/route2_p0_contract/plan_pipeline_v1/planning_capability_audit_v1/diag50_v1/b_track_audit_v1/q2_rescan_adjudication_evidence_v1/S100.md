var MdDBnd = GAUL.filterMetadata('ADM1_NAME', 'equals', 'Madre de Dios');

var LandsatMdD = Landsat.filterBounds(MdDBnd)

                        .filterDate('2020-01-01', '2020-12-31')

                        .filterMetadata('CLOUD_COVER', 'less_than', 80);

function scalingLS(img){

  var optical = img.select('SR_B.').multiply(0.0000275).add(-0.2);

  var thermal = img.select('ST_B.*').multiply(0.00341802).add(149.0);

  return img.addBands(optical, null, true)

              .addBands(thermal, null, true);

}

function maskClouds(img) {

  var cloudShadowBitMask = (1<<4);

  var cloudsBitMask = (1<<3);

  var cloudsCirrusBitMask = (1<<2);

  var cloudsDilatedBitMask = (1<<1);

  var qa = img.select('QA_PIXEL');

  var mask = qa.bitwiseAnd(cloudShadowBitMask).eq(0)

                .and(qa.bitwiseAnd(cloudsBitMask).eq(0))

                .and(qa.bitwiseAnd(cloudsCirrusBitMask).eq(0))

                .and(qa.bitwiseAnd(cloudsDilatedBitMask).eq(0));

  return img.updateMask(mask);

}

function calcVIsLS(img){

  var ndvi = img.normalizedDifference(['SR_B5', 'SR_B4']).select([0], ['ndvi']);

  var lswi = img.normalizedDifference(['SR_B5', 'SR_B6']).select([0], ['lswi']);

  return img.addBands(ndvi).addBands(lswi);

}

var LandsatMdDprepross = LandsatMdD.map(maskClouds).map(scalingLS).map(calcVIsLS);

var medianLS = LandsatMdDprepross.median().clip(MdDBnd).select(['SR_B.','ndvi','lswi']);

var vizParamsLS = {bands: ['SR_B4', 'SR_B3', 'SR_B2'],

                 min: 0,

                 max: 0.15

};

Map.addLayer(medianLS, vizParamsLS, 'median LS');

Map.centerObject(MdDBnd,8);

var chartAgua = ui.Chart.image.regions({

    image: medianLS,

    regions: cuerposDagua,

    scale: 30

});

print(chartAgua);

var chartBosque = ui.Chart.image.regions({

    image: medianLS,

    regions: bosque,

    scale: 30

});

print(chartBosque);

var chartInfra = ui.Chart.image.regions({

    image: medianLS,

    regions: infraestructura,

    scale: 30

});

print(chartInfra);

var puntosColetados = bosque.merge(agPasto).merge(infraestructura).merge(mineria).merge(cuerposDagua);

print('Puntos Coletados',puntosColetados);

Export.table.toDrive({

  collection: puntosColetados,

  description: 'PuntosColetados',

  fileNamePrefix: 'PuntosColetadosMediana2020Landsat',

  fileFormat: 'SHP'

});

var entrenamiento = medianLS

    .sampleRegions({

      collection: puntosColetados,

      properties: ['clase'],

      scale: 30

    });

print('Puntos Entrenamiento',entrenamiento);

var bandasPrediccion = ['SR_B1','SR_B2','SR_B3','SR_B4','SR_B5','SR_B6','SR_B7'];

var CARTclassifier = ee.Classifier.smileCart().train({

  features: entrenamiento,

  classProperty: 'clase', // la propriedad de los puntos

  inputProperties: bandasPrediccion // las bandas para extraer información

});

print('CARTclassifier',CARTclassifier);

var paletaColor = ['47892B', 'F7A708', 'B8B5AE', 'F8FF0D', '0D1AFF']; //https://htmlcolorcodes.com/

var classificacionCART = medianLS.classify(CARTclassifier);

Map.addLayer(classificacionCART, {min: 0, max: 4, palette: paletaColor}, 'clasificación CART',false);

var RFclassifier = ee.Classifier.smileRandomForest({numberOfTrees:10,seed:5}).train({

  features:entrenamiento,

  classProperty: 'clase',

  inputProperties: bandasPrediccion

});

var criteriosRF = RFclassifier.explain();

print('Criterios RF',criteriosRF);

var RFclassificacion = medianLS.classify(RFclassifier);

Map.addLayer(RFclassificacion, {min: 0, max: 4, palette: paletaColor}, 'clasificación RF');

var entrenamientoAgrupador = medianLS.sample({

  region: MdDBnd,

  scale: 30,

  numPixels: 1000

});

var clusterer = ee.Clusterer.wekaKMeans({nClusters:5,seed:5}).train(entrenamientoAgrupador);

var Kclassificacion = medianLS.cluster(clusterer);

Map.addLayer(Kclassificacion.randomVisualizer(), {}, 'clasificación K-means - colores aleatorios',false);

var bandasPrediccionIndices = ['SR_B1','SR_B2','SR_B3','SR_B4','SR_B5','SR_B6','SR_B7','ndvi','lswi'];

var RFclassifierV2 = ee.Classifier.smileRandomForest({numberOfTrees:10,seed:5}).train({

  features:entrenamiento,

  classProperty: 'clase',

  inputProperties: bandasPrediccionIndices

});

var RFclassificacionV2 = medianLS.classify(RFclassifierV2);

Map.addLayer(RFclassificacionV2, {min: 0, max: 4, palette: paletaColor}, 'clasificación RF V2');
