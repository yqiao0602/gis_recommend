  .median()

Map.addLayer(clip, {bands: 'B5,B6,B4'}, 'Terra/Água');

Map.addLayer(clip, {bands: 'B7,B5,B3'}, 'Natural com atmosfera removida', false);

Map.addLayer(clip, {bands: 'B7,B5,B4'}, 'Infravermelho curto', false);

Map.addLayer(clip, {bands: 'B6,B5,B4'}, 'Vegetação', false);

var NDVI = clip.normalizedDifference(['B5', 'B4']).rename('NDVI');

print(NDVI);

Map.addLayer(NDVI, {min: -1, max: 1,palette:['red', 'white', 'green']}, 'NDVI');

var EVI = clip.expression(

  '2.5*((NIR - RED)/(NIR + (6*RED)-(7.5*BLUE) + 1))', {

    'NIR':clip.select('B5'),

    'RED':clip.select('B4'),

    'BLUE':clip.select('B2')}).rename('EVI');

Map.addLayer(EVI, {min: -1, max: 1,palette:['orange', 'white', 'blue']}, 'EVI');

var SAVI = clip.expression(

  '1.5*((NIR - RED)/(NIR + RED + 0.5))', {

    'NIR':clip.select('B5'),

    'RED':clip.select('B4')}).rename('SAVI');

Map.addLayer(SAVI, {min: -1, max: 1,palette:['red', 'white', 'green']}, 'SAVI')

var amostras = floresta.merge(nao_floresta).merge(agua);

var testingPartition = withRandom.filter(ee.Filter.gte('random', split));

var trainedClassifier = ee.Classifier.smileRandomForest(5).train({

  features: trainingPartition,

  classProperty: classe,

  inputProperties: bandas

});

var test = testingPartition.classify(trainedClassifier);

var confusionMatrix = test.errorMatrix(classe, 'classification');

print('Matriz de confusão', confusionMatrix);

var confMatrix = trainedClassifier.confusionMatrix();

print(confMatrix, 'Matriz de confusão');

print(kappa, 'Kappa');

var treinamento = clip.select(bandas).sampleRegions({

  collection: amostras,

  properties: [classe],

  scale: 30,

});

var classificacaoRF = ee.Classifier.smileRandomForest(100).train({

  features: treinamento,

  classProperty: classe,

  inputProperties: bandas

});

var image = FlorRF.select(['classification']);

var areaImage = image.multiply(ee.Image.pixelArea().divide(10000));

var AreaFlorRF = areaImage.reduceRegion({
